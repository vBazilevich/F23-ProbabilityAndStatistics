	\begin{questions}
		\question
		Let \(\xi\) be the number of heads in the experiment of throwing a fair coin twice, and let \(\eta = I(\xi =2)\). Find \(\PP(\xi=x,\eta =y)\) for all suitable values of \(x\) and \(y\).


		\question
		Let \(\xi\) be a Bernoulli random variable with parameter \(p\), and let \(\eta = 1-\xi\), \(\zeta = \xi\eta\). For \(0\leq x, y, z \leq 1\), compute \(\PP(\xi = x, \eta =y)\) and \(\PP(\xi = x, \zeta = z)\).

		\question
		Masha tosses \(n\) coins. Each coin has probability \(p\) of turning up tails and the tosses are independent. Each coin that turns up tails is thrown again. Find the pmf of the number of tails from the second round of throws.


		\question
  		Let \(\xi\) and \(\eta\) be independent random variables, with the same pmf \(p_{\xi}(x) = p_{\eta}(x) = 2^{-x}I(x \in \N)\). Find \(\PP(\min\{\xi,\eta\}\leq x)\), \(\PP(\eta > \xi)\) and \(\PP(\xi \textrm{ divides } \eta)\).
  		
  		\textit{Hint:} $\sum_{i=1}^\infty \frac{1}{2^x - 1} \approx 1.6066$ - Erdos-Borwein constant.


		\question
		Let \(\xi\) and \(\eta\) be independent Poisson r.v., with parameters \(\mu\) and \(\nu\), respectively. Prove that:
		\begin{enumerate}[(a)]
		  \item the r.v.~\(\xi+\eta\) is also Poisson, with parameter \(\mu+\nu\);
		  \item the conditional distribution of \(\xi\) given \(\xi+\eta =n\) is
		  binomial, and find its parameters.
		\end{enumerate}

		\question
		Prove that the sum of two independent binomial variables
with distributions \(\textrm{Bin}(n,p)\) and \(\textrm{Bin}(m,p)\), 
respectively, is \(\textrm{Bin}(n+m,p)\).	
	
		\section*{Problems for self practice}
		\question
		Distribution of random variable $\xi$ is given by $\xi \sim \begin{pmatrix}
			-2 & -1 & 0 & 1 & 2\\
			0.1 & 0.3 & 0.2 & p & 0.2
		\end{pmatrix}$. Determine the probability $p$. Draw the graph of a cumulative distribution function $F_\xi(t)$.
		



		
		\question
		Let us consider the ball of radius $R$ centered at $O$. Point $M$ is chosen at random inside this ball. Random variable $\xi$ is equal to the distance from point $M$ to the sphere. Find the cumulative distribution and probability density functions.
		
		
		
		
		
		\question
		(Walpole 3.5) Determine the value $c$ so that each of the following functions can serve as a probability distribution of the discrete random variable $X$.
		
		\begin{parts}
			\part $f(x) = c(x^2 + 4)$, $x \in \{0, 1, 2, 3\}$;
			\part $f(x) = c \binom{2}{x}\binom{3}{3-x}$, $x\in \{0, 1, 2\}$.
		\end{parts}
		




		
		\question
		(Walpole 3.35) Suppose it is known from large amounts of historical data that $X$, the number of cars that arrive at a specific intersection during a 20-second time period, is characterized by the following discrete probability function: $f(x) = e^{-6}\frac{6^x}{x!}$, for $x = 0, 1, 2, \dots$.
		
		\begin{parts}
			\part Find the probability that in a specific 20-second
			time period, more than 8 cars arrive at the
			intersection.
			\part Find the probability that only 2 cars arrive.
		\end{parts}
		
		\textit{Hint:} $e^6 \approx 400$.
		



		
		
		\question
		Consider a system of components in which there are 5 independent components, each of which possesses an operational probability of 0.92. The system does have a redundancy built in such that it does not fail if 3 out of the 5 components are operational.
		
		\begin{parts}
			\part Write down the probability mass function;
			\part What is the probability that the total system is operational?
		\end{parts}
		




		
 	\end{questions}
 	
 	
 	
 	
 	\pagebreak
 	\section*{Answers}
 	\begin{enumerate}
 	
 		\setcounter{enumi}{6}
 		\item $p = 0.2$.
 		\item $f_\xi(x) = \begin{cases}\frac{3 (R - x) ^ 2}{R^3}, \quad 0 \leq 0 \leq R \\ 0, \quad \textit{otherwise}\end{cases}$; $F_\xi(x) = \begin{cases}0, \quad x \leq 0 \\ \frac{(R - x) ^ 3}{R^3}, \quad 0 \leq 0 \leq R \\ 1, \quad \textit{otherwise}\end{cases}$
 		\item (a) $\frac{1}{30}$, (b) $\frac{1}{10}$.
 		\item (a) $0.1528$ (b) $0.0446$
 		\item (a) $f(x) = \begin{cases}
			\binom{5}{x} \cdot 0.92^x \cdot (1 - 0.92)^{5 - x},\, x \in \{0, 1, 2, 3, 4, 5\}\\
			0,\, \text{otherwise}
		\end{cases}$; (b) 0.9955
 	\end{enumerate}
