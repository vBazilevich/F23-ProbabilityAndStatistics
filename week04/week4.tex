		\question
		Find \(\EE \xi(\xi-1)\), given that \(\xi\sim \textrm{Poi}(\lambda)\). Then find \(\Var\ \xi\). 



		\question
		Show that \(\Var\) is insensitive to shifts, that is: \(\Var\ \xi = \Var(\xi +c)\) for any constant \(c\) and random variable \(\xi\) for which \(\Var\ \xi\) exists.


		\question
		Find \(\EE\xi\) and \(\Var\ \xi\), if the random variable \(\xi\) has distribution function (cdf) \[ F(x) = x I\big(\frac{1}{3}\leq x < \frac{1}{2}\big) + 
I\big(x \geq \frac{1}{2}\big). \]


		\question
  		The joint distribution of the random vector \((\xi_1,\xi_2)\) is determined by the conditions: \(\PP(\xi_1 \xi_2 =0) = 1\), \(\PP(\xi_j=1)=\PP(\xi_j=-1) = \frac{1}{4}\) for \(j=1,2\).  Find \(\EE\xi_1\), \(\EE\xi_2\), \(\Var\ \xi_1\), \(\Var\ \xi_2\), \(\textrm{Cov}\ (\xi_1,\xi_2)\).

		\question
		A total of \(N\) dipopers were manufactured at Lobanov factory in central Choknutistan, with each dipoper having a defect with probability \(p\). Each dipoper undergoes a test which detects the defect (if present) with probability \(r\). Let \(\xi\) be the number of defective
dipopers, \(\eta\) the number of dipopers that were revealed to have a defect by the test. Assuming that each dipoper is defective or not independently of all the others, and that the outcome of each test is also independent of all the other outcomes, prove that \[ \EE[\xi | \eta] = \frac{Np(1-r)+(1-p)\eta}{1-pr}. \]


		\question
		Find the mean (expected value) and variance of a geometric random variable with parameter \(p\).


		\question
		Let the random variable \(\xi\) be the number of elements \((1,2,\ldots,n)\) that stay fixed after a random permutation. Find \(\EE\xi\) and \(\Var\ \xi\).	
	
	\section*{Tasks for homework}
		\question
		Random variable $\zeta$ can take values $-1$, $0$ and $1$. Find its probability mass function given that $\EE\zeta = 0$, $\Var \zeta = 0.5$.
		





		
		\question
		The cards are drawn at random one after another from a deck of 52 cards (and they are not returned back to the deck). One continues to draw until he meets the ace of diamonds. What is the average number of cards drawn from the deck? Find the probability that one does not need to take more than half of all of the cards.
		






		
		\question
		The probabilities to pass the exam for three students are equal to $0.9$, $0.8$ and $0.7$ respectively. They pass or fail independently from each other. Let $\xi$ be equal to the number of students who have passed the exam. Find $\EE\xi$ and $\Var\xi$.
		





		
		\question
		Eight random balls are taken out of the urn that contains 10 white balls and 15 black balls. What is the average number of white balls among the ones taken?
		
		
		
		
		
		
		\question
		Ten people entered the lift at the ground floor of a nine-storey building. How many stops is the lift going to make as it goes up?
		




		
		\question
		One of three-digit numbers is chosen at random. Determine the expected value of the quantity of different even digits used to write down this number.
		




		
		\question
		43 equally strong sportsmen take part in a ski race; 18 of them belong to club A, 10 to club B and 15 to club C. What is the average place for:
		
		\begin{parts}
			\part the best participant from club B;
			\part the worst participant from club B.
		\end{parts}
		




		
		\question
		Let $Y$ be the number of sixes and $Z$ be the number of fours one gets when rolling six dice. Find the expected value and variance of $Y + Z$.
		




		
		\question

			$2$ & $\frac{1}{17}$ & $\frac{5}{17}$ & $\frac{3}{17}$\\\hline
		\end{tabular}

			\part Determine if $\xi$ and $\eta$ are independent;
			\part Find covariance of $\xi$ and $\eta$;
			\part Find conditional expected values 
		\end{parts}

			Let $Y$ be the number of times one needs to flip a coin to get tails and heads at least once. Find the expected value of $Y$.
